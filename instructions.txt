    if test -d $dir
        python3 scripts/06_collect_state.py --raw-dir $dir
    end
end

# Для bash
for dir in data/raw/*; do
  if [ -d "$dir" ]; then
    python3 scripts/06_collect_state.py --raw-dir "$dir"
  fi
done


# Расчёт метрик
# Для fish
for dir in data/raw/*
    if test -d $dir
        set experiment_name (basename $dir)

        python3 scripts/07_calculate_metrics.py \
            --input-dir $dir \
            --output-dir data/processed/$experiment_name
    end
end

# Для bash
for dir in data/raw/*; do
  if [ -d "$dir" ]; then
    experiment_name="$(basename "$dir")"

    python3 scripts/07_calculate_metrics.py \
      --input-dir "$dir" \
      --output-dir "data/processed/$experiment_name"
  fi
done
